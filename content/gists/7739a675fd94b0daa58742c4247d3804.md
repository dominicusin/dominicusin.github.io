## dochroot

```Shell
#!/bin/bash

echo $1

for i in dev sys proc run 
do
  mount -R /$i $1/$i
  mount --make-rslave  $1/$i
done 

chroot $1

umount -l  `mount|grep $1/|cut -d' ' -f3`

```
