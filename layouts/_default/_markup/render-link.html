{{/* Trust-boundary render-link hook (overrides themes/blowfish/.../render-link.html). Mirrors Blowfish's urls.Parse-based pass-through EXACTLY, but wraps the parse in `try` so that malformed destinations (e.g. README links containing `[...]` like the awesome-mac App Store URL) degrade to plain text instead of aborting the whole build. Additionally neutralises dangerous link schemes (javascript:/data:/vbscript:/file:) by rendering as plain text. */}} {{- $dest := .Destination -}} {{- $parsed := try (urls.Parse $dest) -}} {{- if $parsed.Err -}} {{- /* malformed URL: render as text, don't abort the build */ -}} {{- .Text | safeHTML -}} {{- else -}} {{- $u := $parsed.Value -}} {{- $href := $u.String -}} {{- $lower := lower $dest -}} {{- $dangerous := or (hasPrefix $lower "javascript:") (hasPrefix $lower "data:") (hasPrefix $lower "vbscript:") (hasPrefix $lower "file:") -}} {{- if $dangerous -}} {{- /* neutralise executable/unsafe schemes: render as text, no anchor */ -}} {{- .Text | safeHTML -}} {{- else -}} {{- if strings.HasPrefix $href "#" -}} {{- $href = printf "%s#%s" .PageInner.RelPermalink $u.Fragment -}} {{- else if and $href (not $u.IsAbs) -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- with or ($.PageInner.GetPage $path) ($.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- $href = .RelPermalink -}} {{- with $u.RawQuery -}}{{- $href = printf "%s?%s" $href . -}}{{- end -}} {{- with $u.Fragment -}}{{- $href = printf "%s#%s" $href . -}}{{- end -}} {{- end -}} {{- end -}} {{- $forceNewTab := .Page.Params.externalLinkForceNewTab | default (.Page.Site.Params.article.externalLinkForceNewTab | default true) -}} {{- $isExternal := or (strings.HasPrefix $dest "http:") (strings.HasPrefix $dest "https:") -}} {{- $newTab := and $forceNewTab $isExternal -}} {{- .Text | safeHTML -}} {{- end -}} {{- end -}}
